# 第一部分：场景互动总览

## 一、游戏状态机

### 1.1 状态定义

游戏使用状态机管理场景切换，确保游戏流程的有序进行。

```
┌─────────────────────────────────────────────────────────────────┐
│                        游戏状态机                                │
└─────────────────────────────────────────────────────────────────┘

    ┌──────────┐
    │   BOOT   │  游戏启动初始化
    └─────┬────┘
          │
          ▼
    ┌──────────┐
    │  TITLE   │  标题界面，主菜单
    └─────┬────┘
          │
          ▼
    ┌──────────┐
    │   BASE   │  基地场景，准备阶段
    └─────┬────┘
          │
          ▼
    ┌──────────┐
    │MAP_SELECT│  地图选择界面
    └─────┬────┘
          │
          ▼
    ┌──────────┐
    │EXPEDITION│  远征地图场景
    └─────┬────┘
          │
     ┌────┴────┐
     │         │
     ▼         ▼
┌─────────┐ ┌─────────┐
│ BATTLE  │ │ RESULT  │
└────┬────┘ └────┬────┘
     │           │
     └─────┬─────┘
           │
           ▼
    ┌──────────┐
    │GAME_OVER │  游戏结束/通关
    └──────────┘
```

### 1.2 状态切换触发条件

| 当前状态 | 目标状态 | 触发条件 | 数据传递 |
|:---|:---|:---|:---|
| BOOT | TITLE | 初始化完成 | 无 |
| TITLE | BASE | 点击继续游戏/新游戏 | 加载存档或创建新存档 |
| BASE | MAP_SELECT | 点击指挥中心 | 传递玩家当前资源 |
| MAP_SELECT | EXPEDITION | 确认地图和难度 | 传递 map_id, difficulty |
| EXPEDITION | BATTLE | 进入战斗节点 | 传递 enemy_config, environment |
| BATTLE | RESULT | 战斗结束（胜/负） | 传递 battle_result, rewards |
| RESULT | EXPEDITION | 继续远征 | 更新远征进度 |
| RESULT | BASE | 返回基地 | 保存远征进度，发放奖励 |
| EXPEDITION | GAME_OVER | 玩家 HP≤0 或通关 | 传递 final_result |

### 2.2 核心数据结构

#### PlayerData（玩家数据）

```gdscript
class_name PlayerData extends RefCounted

# 基础信息
var player_id: String          # 玩家唯一 ID
var character_id: String       # 当前角色 ID
var character_data: Character  # 角色实例

# 资源
var stardust: int              # 星尘
var star_core: int             # 星核
var gold: int                  # 金币

# 解锁内容
var unlocked_characters: Array[String]  # 已解锁角色
var unlocked_cards: Dictionary          # 已解锁卡牌 {card_id: count}
var unlocked_flows: Dictionary          # 流派等级 {flow_id: level}

# 收藏
var collected_cards: Array[CardInstance]  # 收藏卡牌
var collected_affixes: Array[String]      # 已解锁词条

# 成就
var achievements: Dictionary  # {achievement_id: completed}

# 统计
var stats: PlayerStats  # 游戏统计数据

# 当前远征
var current_expedition: ExpeditionData
```

#### ExpeditionData（远征数据）

```gdscript
class_name ExpeditionData extends RefCounted

var map_id: String           # 地图 ID
var difficulty: String       # 难度等级
var current_node_id: String  # 当前节点 ID
var visited_nodes: Array[String]  # 已访问节点
var battle_wins: int         # 胜利次数
var battle_losses: int       # 失败次数
var rewards: Dictionary      # 已获得奖励
var map_progress: float      # 地图进度 0-100%
```

#### BattleData（战斗数据）

```gdscript
class_name BattleData extends RefCounted

var enemies: Array[EnemyData]    # 敌人列表
var environment: EnvironmentData # 当前环境
var player_deck: Deck            # 玩家卡组

var player_discard: Pile         # 弃牌堆
var turn_number: int             # 当前回合
var battle_result: BattleResult  # 战斗结果
```

## 三、信号系统

### 3.1 全局信号总线

```gdscript
# GameEvents.gd - 全局信号单例
extends Node

# 游戏状态信号
signal game_state_changed(old_state, new_state)
signal scene_loaded(scene_name)

# 玩家数据信号
signal player_resources_changed(stardust, star_core, gold)
signal player_hp_changed(current, max)
signal character_unlocked(character_id)

# 战斗信号
signal battle_started(enemies, environment)
signal turn_started(turn_number)
signal turn_ended(turn_number)
signal card_played(card_data)
signal enemy_action_taken(enemy_id, action)
signal battle_won(rewards)
signal battle_lost(penalty)

# 成就信号
signal achievement_unlocked(achievement_id)
signal achievement_progress(achievement_id, current, target)

# UI 信号
signal ui_show_popup(title, description)
signal ui_show_reward(reward_data)
signal ui_transition_started()
signal ui_transition_completed()
```

### 3.2 信号连接示例

```gdscript
# 在 GameManager 中连接信号
func _ready():
    BattleManager.battle_won.connect(_on_battle_won)
    BattleManager.battle_lost.connect(_on_battle_lost)
    AchievementManager.achievement_unlocked.connect(_on_achievement_unlocked)

func _on_battle_won(rewards: Dictionary):
    # 发放奖励
    player_data.stardust += rewards.get("stardust", 0)
    player_data.star_core += rewards.get("star_core", 0)
    
    # 通知 UI
    GameEvents.ui_show_reward.emit(rewards)
    
    # 检查成就
    AchievementManager.check_achievements("battle_win", {})

func _on_battle_lost(penalty: Dictionary):
    # 扣除生命
    var hp_loss = penalty.get("hp_loss", 10)
    player_data.character_data.current_hp -= hp_loss
    
    # 检查游戏结束
    if player_data.character_data.current_hp <= 0:
        change_state(GameState.GAME_OVER)
```

#### 升级逻辑代码

```gdscript
# BuildingManager.gd
func upgrade_forge_workshop() -> bool:
    var current_level = buildings["forge_workshop"]
    if current_level >= 10:
        return false  # 已满级
    
    var cost = get_forge_upgrade_cost(current_level)
    
    if cost.type == "stardust":
        if player_data.stardust < cost.amount:
            return false  # 资源不足
        player_data.stardust -= cost.amount
    else:
        if player_data.star_core < cost.amount:
            return false
        player_data.star_core -= cost.amount
    
    buildings["forge_workshop"] += 1
    
    # 更新系数
    K_craft = 1.0 - (buildings["forge_workshop"] * 0.03)
    K_craft = max(0.7, K_craft)
    
    # 保存并通知
    SaveManager.save_building_data(buildings)
    GameEvents.player_resources_changed.emit(
        player_data.stardust, 
        player_data.star_core, 
        player_data.gold
    )
    
    return true
```

### 1.2 节点生成规则

```gdscript
func _generate_nodes(map_config: MapConfig):
    nodes.clear()
    
    var node_count = map_config.node_count  # 通常 10-15 个节点
    
    for i in range(node_count):
        var node = MapNode.new()
        node.id = "node_%d" % i
        node.index = i
        
        # 确定节点类型
        if i == 0:
            node.type = NodeType.START  # 起点
        elif i == node_count - 1:
            node.type = NodeType.BOSS   # BOSS
        else:
            node.type = _get_random_node_type(i, map_config)
        
        # 绑定环境
        node.environment = _get_environment_for_node(i, map_config)
        
        # 配置敌人/奖励
        _configure_node_content(node, i, map_config)
        
        nodes.append(node)
```

```gdscript
func calculate_battle_rewards(enemy_tier: int, difficulty: String) -> Dictionary:
    var base_stardust = 25 * enemy_tier
    var base_star_core = 0
    
    if enemy_tier >= 5:  # 精英有星核
        base_star_core = 8
    
    # 应用系数
    var K_diff = get_difficulty_resource_multiplier(difficulty)
    var K_map = get_map_resource_multiplier(expedition_data.map_id)
    var K_refine = get_refine_multiplier()
    
    return {
        "stardust": floor(base_stardust * K_diff * K_map * K_refine),
        "star_core": floor(base_star_core * K_diff * K_map * K_refine)
    }
```

### 3.1 路径连接规则


### 3.2 移动逻辑

```gdscript
func move_to_node(target_node_id: String) -> bool:
    var current_node = get_node(expedition_data.current_node_id)
    var target_node = get_node(target_node_id)
    
    # 检查是否是相邻节点
    if not current_node.connected_nodes.has(target_node_id):
        show_error("无法移动到该节点")
        return false
    
    # 更新位置
    expedition_data.current_node_id = target_node_id
    expedition_data.visited_nodes.append(target_node_id)
    
    # 更新进度
    expedition_data.map_progress = calculate_progress()
    
    # 触发节点事件
    _trigger_node_event(target_node)
    
    return true
```

## 四、伤害计算完整流程

### 4.1 伤害计算公式

```
┌─────────────────────────────────────────────────────────────────┐
│                        伤害计算流程                              │
└─────────────────────────────────────────────────────────────────┘

卡牌基础伤害 (Base Damage)
         │
         ▼
    + 状态加成 (Status Bonus)
      例如：强化×5, 机关层数×3
         │
         ▼
    = 基础伤害 (Base Total)
         │
         ▼
    × 环境系数 (K_env)
      例如：雨天电系 1.5, 高温火系 1.3
         │
         ▼
    = 环境修正伤害 (Environment Modified)
         │
         ▼
    × 暴击修正 (Crit Modifier)
      (1 + 暴击率 × (暴击倍率 - 1))
         │
         ▼
    = 暴击修正伤害 (Crit Modified)
         │
         ▼
    × 防御修正 (K_def)
      (1 - 防御力/(防御力 + 100))
         │
         ▼
    = 最终伤害 (Final Damage)
      最低为 1
```

### 4.2 伤害计算代码

```gdscript
# DamageCalculator.gd
static func calculate_damage(
    base_damage: int,
    element: String,
    attacker: Character,
    target: Enemy,
    environment: EnvironmentData
) -> int:
    # 1. 状态加成
    var status_bonus = _calculate_status_bonus(attacker, element)
    var total = base_damage + status_bonus
    
    # 2. 环境修正
    var env_multiplier = environment.get_damage_multiplier(element)
    total = int(total * env_multiplier)
    
    # 3. 暴击计算
    var crit_rate = attacker.get_crit_rate()
    var crit_mult = attacker.get_crit_multiplier()
    var is_crit = RandomUtil.check_crit(crit_rate)
    
    if is_crit:
        total = int(total * crit_mult)
        GameEvents.crit_triggered.emit()
    
    # 4. 防御修正
    var def_multiplier = 1.0 - (target.defense / float(target.defense + 100))
    total = int(total * def_multiplier)
    
    # 5. 最低伤害
    return max(1, total)

static func _calculate_status_bonus(attacker: Character, element: String) -> int:
    var bonus = 0
    
    # 强化状态
    if attacker.has_effect("strengthen"):
        bonus += attacker.get_effect_stacks("strengthen") * 5
    
    # 流派专属状态
    match attacker.current_flow:
        "mechanism":
            if attacker.has_effect("mechanism"):
                bonus += attacker.get_effect_stacks("mechanism") * 3
        "steam":
            if attacker.has_effect("steam_pressure"):
                bonus += attacker.get_effect_stacks("steam_pressure") * 2
    
    return bonus
```

```gdscript
var player_effects: Dictionary = {}  # {effect_id: stacks}

func add_effect(target: String, effect_id: String, stacks: int = 1):
    var effects_dict = _get_effects_dict(target)
    
    if effect_id not in effects_dict:
        effects_dict[effect_id] = 0
    
    effects_dict[effect_id] += stacks
    
    # 检查上限
    var max_stacks = get_effect_max_stacks(effect_id)
    effects_dict[effect_id] = min(effects_dict[effect_id], max_stacks)
    
    # 通知 UI
    GameEvents.effect_applied.emit(target, effect_id, effects_dict[effect_id])

func remove_effect(target: String, effect_id: String, stacks: int = 1):
    var effects_dict = _get_effects_dict(target)
    
    if effect_id in effects_dict:
        effects_dict[effect_id] -= stacks
        if effects_dict[effect_id] <= 0:
            effects_dict.erase(effect_id)
        
        GameEvents.effect_removed.emit(target, effect_id)

func resolve_dot_effects():
    # 结算持续伤害效果
    for effect_id in player_effects:
        var damage = get_dot_damage(effect_id, player_effects[effect_id])
        if damage > 0:
            player_data.character.take_damage(damage)
            show_damage_number("player", damage)
```

```gdscript
func generate_card_options(count: int) -> Array[Card]:
    var options = []
    var card_pool = _get_available_card_pool()
    
    for i in range(count):
        # 权重抽取（稀有度权重）
        var card = _weighted_random_card(card_pool)
        options.append(card)
    
    # 确保选项多样性
    options = _ensure_diversity(options)
    
    return options

func _weighted_random_card(card_pool: Array) -> Card:
    var weights = []
    for card in card_pool:
        # 稀有度越低权重越高
        var weight = 10 - card.tier * 1.2
        weights.append(max(1, weight))
    
    return card_pool.pick_random_with_weights(weights)
```

```gdscript
# AchievementManager.gd
func check_achievements(trigger_type: String, data: Dictionary):
    for achievement in achievement_resources:
        if achievement.condition_type != trigger_type:
            continue
        
        if is_achievement_completed(achievement, data):
            unlock_achievement(achievement.achievement_id)
        else:
            update_progress(achievement.achievement_id, data.get("progress", 1))

func is_achievement_completed(achievement: AchievementResource, data: Dictionary) -> bool:
    var current_progress = progress.get(achievement.achievement_id, 0)
    return current_progress >= achievement.condition_value
```

```gdscript
        }
```

## 二、资源获取公式应用

### 2.1 完整资源计算

```gdscript
func calculate_total_resources() -> Dictionary:
    var total_stardust = 0
    var total_star_core = 0
    
    # 战斗奖励累加
    for battle in expedition_battles:
        total_stardust += battle.stardust_reward
        total_star_core += battle.star_core_reward
    
    # 应用精炼系数
    var K_refine = BuildingManager.get_refine_multiplier()
    total_stardust = floor(total_stardust * K_refine)
    total_star_core = floor(total_star_core * K_refine)
    
    # 通关奖励
    if expedition_completed:
        var clear_rewards = calculate_expedition_clear_rewards()
        total_stardust += clear_rewards.stardust
        total_star_core += clear_rewards.star_core
    
    return {
        "stardust": total_stardust,
        "star_core": total_star_core
    }
```

### 4.1 战斗主循环

```gdscript
# BattleManager.gd
func battle_loop():
    while battle_state not in [VICTORY, DEFEAT]:
        # 玩家回合
        start_player_turn()
        
        # 等待玩家行动
        while battle_state == PLAYER_TURN:
            await input_received
        
        # 敌人回合
        start_enemy_turn()
        
        # 敌人行动
        for enemy in enemies:
            await execute_enemy_action(enemy)
        
        # 状态结算
        await resolve_end_of_turn_effects()
        
        # 检查结束条件
        if check_victory():
            battle_state = VICTORY
        elif check_defeat():
            battle_state = DEFEAT
    
    # 战斗结束
    if battle_state == VICTORY:
        on_battle_won()
    else:
        on_battle_lost()
```

```gdscript
# SaveManager.gd
func save_game(slot: int) -> bool:
    var save_data = {
        "version": GAME_VERSION,
        "timestamp": Time.get_unix_time_from_system(),
        "player": player_data.to_dict(),
        "buildings": BuildingManager.buildings,
        "achievements": AchievementManager.achievements,
        "settings": GameSettings.to_dict()
    }
    
    var file_path = SAVE_PATH + "save_%d.save" % slot
    var file = FileAccess.open(file_path, FileAccess.WRITE)
    
    if file:
        file.store_var(save_data)
        file.close()
        return true
    
    return false

func load_game(slot: int) -> Dictionary:
    var file_path = SAVE_PATH + "save_%d.save" % slot
    
    if not FileAccess.file_exists(file_path):
        return {}
    
    var file = FileAccess.open(file_path, FileAccess.READ)
    var save_data = file.get_var()
    file.close()
    
    # 版本迁移
    if save_data.get("version", 0) < GAME_VERSION:
        save_data = migrate_save_data(save_data)
    
    return save_data
```

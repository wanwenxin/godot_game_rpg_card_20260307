### 一、游戏管理器 (GameManager)

**职责**：全局游戏状态管理、场景切换、数据传递

```gdscript
# scripts/autoload/GameManager.gd
extends Node

signal game_state_changed(old_state: GameState, new_state: GameState)
signal scene_transition_started()
signal scene_transition_completed()

enum GameState {
    BOOT,           # 启动
    TITLE,          # 标题界面
    BASE,           # 基地
    MAP_SELECT,     # 地图选择
    EXPEDITION,     # 远征中
    BATTLE,         # 战斗中
    RESULT,         # 结果界面
    GAME_OVER       # 游戏结束
}

var current_state: GameState = GameState.BOOT
var player_data: PlayerData
var battle_data: BattleData
var expedition_data: ExpeditionData

# 游戏设置
var settings: GameSettings

func _ready():
    _init_game()

func _init_game():
    # 初始化游戏
    settings = GameSettings.new()
    settings.load()
    player_data = PlayerData.new()
    _load_save_data()

func change_state(new_state: GameState):
    var old_state = current_state
    current_state = new_state
    game_state_changed.emit(old_state, new_state)
    _on_state_changed(old_state, new_state)

func _on_state_changed(old_state: GameState, new_state: GameState):
    match new_state:
        GameState.TITLE:
            _load_title_scene()
        GameState.BASE:
            _load_base_scene()
        GameState.MAP_SELECT:
            _load_map_select_scene()
        GameState.BATTLE:
            _load_battle_scene()

func start_expedition(map_id: String, difficulty: String):
    expedition_data = ExpeditionData.new()
    expedition_data.map_id = map_id
    expedition_data.difficulty = difficulty
    expedition_data.start()
    change_state(GameState.EXPEDITION)

func start_battle(enemies: Array, rewards: Dictionary):
    battle_data = BattleData.new()
    battle_data.enemies = enemies
    battle_data.rewards = rewards
    change_state(GameState.BATTLE)

func _load_title_scene():
    await get_tree().change_scene_to_file("res://scenes/main/TitleScene.tscn")
    scene_transition_completed.emit()

func _load_base_scene():
    await get_tree().change_scene_to_file("res://scenes/game/BaseScene.tscn")
    scene_transition_completed.emit()

func _load_map_select_scene():
    await get_tree().change_scene_to_file("res://scenes/game/MapSelectScene.tscn")
    scene_transition_completed.emit()

func _load_battle_scene():
    await get_tree().change_scene_to_file("res://scenes/game/BattleScene.tscn")
    scene_transition_completed.emit()
```

### 二、战斗管理器 (BattleManager)

**职责**：战斗流程控制、回合管理、卡牌使用、敌人 AI

```gdscript
# scripts/autoload/BattleManager.gd
extends Node

signal battle_started()
signal turn_started(turn_number: int)
signal turn_ended(turn_number: int)
signal player_energy_changed(current: int, max: int)
signal card_played(card_data: CardData)
signal enemy_intent_changed(enemy_id: String, intent: EnemyIntent)
signal battle_won()
signal battle_lost()

# 战斗状态
enum BattleState {
    WAITING,        # 等待开始
    PLAYER_TURN,    # 玩家回合
    ENEMY_TURN,     # 敌人回合
    RESOLVING,      # 效果结算中
    VICTORY,        # 胜利
    DEFEAT          # 失败
}

var battle_state: BattleState = BattleState.WAITING
var turn_number: int = 0
var player_energy: int = 3
var max_energy: int = 3

# 战斗实体
var player_deck: Deck
var player_hand: Hand
var player_discard: Pile
var player_draw: Pile
var enemies: Array[Enemy] = []

# 状态效果
var player_effects: Dictionary = {}
var enemy_effects: Dictionary = {}

func _ready():
    pass

func start_battle(battle_data: BattleData):
    battle_state = BattleState.WAITING
    
    # 初始化战斗实体
    _init_deck()
    _init_enemies(battle_data.enemies)
    _init_effects()
    
    # 开始战斗
    turn_number = 0
    player_energy = max_energy
    battle_state = BattleState.PLAYER_TURN
    
    # 抽初始手牌
    player_hand.draw_cards(5)
    
    battle_started.emit()
    start_player_turn()

func _init_deck():
    player_deck = Deck.new()
    player_hand = Hand.new()
    player_discard = Pile.new()
    player_draw = Pile.new()
    
    # 根据角色和流派构建卡组
    var character = GameManager.player_data.current_character
    var flow = character.current_flow
    player_deck.build_from_flow(flow)

func _init_enemies(enemies_data: Array):
    enemies.clear()
    for enemy_data in enemies_data:
        var enemy = Enemy.new()
        enemy.setup(enemy_data)
        enemies.append(enemy)

func _init_effects():
    player_effects.clear()
    enemy_effects.clear()

# 玩家回合
func start_player_turn():
    battle_state = BattleState.PLAYER_TURN
    turn_number += 1
    
    # 重置能量
    player_energy = max_energy
    player_energy_changed.emit(player_energy, max_energy)
    
    # 重置敌人意图
    for enemy in enemies:
        enemy.reset_intent()
    
    # 抽牌
    player_hand.draw_cards(3)
    
    # 更新敌人意图显示
    _update_enemy_intents()
    
    turn_started.emit(turn_number)

func _update_enemy_intents():
    for enemy in enemies:
        var intent = enemy.get_next_intent()
        enemy_intent_changed.emit(enemy.id, intent)

func play_card(card: Card):
    if battle_state != BattleState.PLAYER_TURN:
        return false
    
    if player_energy < card.cost:
        return false
    
    # 消耗能量
    player_energy -= card.cost
    player_energy_changed.emit(player_energy, max_energy)
    
    # 执行卡牌效果
    card.execute()
    
    # 移动卡牌到弃牌堆
    player_hand.remove_card(card)
    player_discard.add_card(card)
    
    card_played.emit(card.data)
    
    # 检查是否所有敌人死亡
    if _check_victory():
        battle_state = BattleState.VICTORY
        battle_won.emit()
    
    return true

func end_player_turn():
    if battle_state != BattleState.PLAYER_TURN:
        return
    
    battle_state = BattleState.ENEMY_TURN
    turn_ended.emit(turn_number)
    
    # 敌人行动
    await _execute_enemy_turn()
    
    # 检查是否失败
    if _check_defeat():
        battle_state = BattleState.DEFEAT
        battle_lost.emit()
        return
    
    # 开始下回合
    start_player_turn()

func _execute_enemy_turn():
    battle_state = BattleState.RESOLVING
    
    for enemy in enemies:
        if enemy.is_alive():
            await enemy.execute_intent()
            await get_tree().create_timer(0.5).timeout
    
    battle_state = BattleState.PLAYER_TURN

func _check_victory() -> bool:
    for enemy in enemies:
        if enemy.is_alive():
            return false
    return true

func _check_defeat() -> bool:
    return GameManager.player_data.current_hp <= 0

# 状态效果管理
func add_effect(target: String, effect_id: String, stacks: int = 1):
    var effects_dict = player_effects if target == "player" else enemy_effects
    if effect_id not in effects_dict:
        effects_dict[effect_id] = 0
    effects_dict[effect_id] += stacks

func remove_effect(target: String, effect_id: String, stacks: int = 1):
    var effects_dict = player_effects if target == "player" else enemy_effects
    if effect_id in effects_dict:
        effects_dict[effect_id] -= stacks
        if effects_dict[effect_id] <= 0:
            effects_dict.erase(effect_id)

func get_effect_stacks(target: String, effect_id: String) -> int:
    var effects_dict = player_effects if target == "player" else enemy_effects
    return effects_dict.get(effect_id, 0)
```

### 三、卡牌数据库 (CardDatabase)

**职责**：卡牌数据加载、查询、缓存

```gdscript
# scripts/autoload/CardDatabase.gd
extends Node

var card_data_cache: Dictionary = {}
var flow_cards: Dictionary = {}  # 按流派分类的卡牌

func _ready():
    _load_all_cards()

func _load_all_cards():
    # 加载所有卡牌配置
    _load_common_cards()
    _load_flow_cards()

func _load_common_cards():
    var dir = DirAccess.open("res://resources/cards/common/")
    if dir:
        dir.list_dir_begin()
        var file_name = dir.get_next()
        while file_name != "":
            if file_name.ends_with(".tres"):
                var card_data = load("res://resources/cards/common/" + file_name)
                card_data_cache[card_data.card_id] = card_data
            file_name = dir.get_next()

func _load_flow_cards():
    var flows = ["puppet", "trap", "mechanism", "steam", "machinery", 
                 "lightning", "holy_light", "guard", "punishment",
                 "shooting", "nature", "stealth", "element", "life", 
                 "matter", "sun", "moon", "star", "body", "qi", "blood"]
    
    for flow in flows:
        flow_cards[flow] = []
        var dir_path = "res://resources/cards/flows/%s/" % flow
        var dir = DirAccess.open(dir_path)
        if dir:
            dir.list_dir_begin()
            var file_name = dir.get_next()
            while file_name != "":
                if file_name.ends_with(".tres"):
                    var card_data = load(dir_path + file_name)
                    card_data_cache[card_data.card_id] = card_data
                    flow_cards[flow].append(card_data.card_id)
                file_name = dir.get_next()

func get_card_data(card_id: String) -> CardData:
    if card_id in card_data_cache:
        return card_data_cache[card_id]
    return null

func get_cards_by_flow(flow_id: String) -> Array:
    return flow_cards.get(flow_id, [])

func get_cards_by_type(card_type: CardData.CardType) -> Array:
    var result = []
    for card_id in card_data_cache:
        var data = card_data_cache[card_id]
        if data.type == card_type:
            result.append(card_id)
    return result

func create_card_instance(card_id: String, tier: int = 1) -> Card:
    var data = get_card_data(card_id)
    if data:
        var card = Card.new()
        card.setup(data, tier)
        return card
    return null
```

### 四、卡牌系统 (Card)

**职责**：卡牌数据、效果执行、UI 显示

```gdscript
# scripts/cards/Card.gd
extends Node
class_name Card

signal card_played(card: Card)

var data: CardData
var tier: int = 1
var current_cost: int = 0
var target: Node = null

# 卡牌实例 ID（用于区分同名卡牌）
var instance_id: String = ""

# 附加词条
var affixes: Array = []

func setup(card_data: CardData, card_tier: int = 1):
    data = card_data
    tier = card_tier
    current_cost = _calculate_cost()

func _calculate_cost() -> int:
    # 根据品级计算消耗
    var base_cost = data.get_cost_for_tier(tier)
    
    # 词条修正
    for affix in affixes:
        if affix == "cost_down":
            base_cost -= 1
        elif affix == "cost_up":
            base_cost += 1
    
    return max(0, base_cost)

func execute():
    # 执行卡牌效果
    match data.type:
        CardData.CardType.ATTACK:
            _execute_attack()
        CardData.CardType.SKILL:
            _execute_skill()
        CardData.CardType.STATUS:
            _execute_status()
        CardData.CardType.CORE:
            _execute_core()
    
    card_played.emit(self)

func _execute_attack():
    var damage = _calculate_damage()
    
    # 应用伤害
    if target and target is Enemy:
        target.take_damage(damage, data.element)
        
        # 触发额外效果
        _apply_attack_effects()

func _calculate_damage() -> int:
    var base_damage = data.get_damage_for_tier(tier)
    
    # 状态加成
    var effect_bonus = BattleManager.calculate_effect_bonus(data.element)
    
    # 环境修正
    var env_bonus = BattleManager.calculate_environment_bonus(data.element)
    
    # 暴击计算
    var crit_rate = GameManager.player_data.get_crit_rate()
    var crit_mult = GameManager.player_data.get_crit_multiplier()
    var is_crit = randf() < crit_rate
    
    var total_damage = base_damage + effect_bonus + env_bonus
    if is_crit:
        total_damage = int(total_damage * crit_mult)
    
    return total_damage

func _apply_attack_effects():
    # 施加状态效果
    if data.apply_effects:
        for effect in data.apply_effects:
            BattleManager.add_effect("enemy", effect.id, effect.stacks)

func _execute_skill():
    # 执行技能效果
    match data.effect_id:
        "draw":
            player_hand.draw_cards(data.effect_value)
        "energy":
            BattleManager.player_energy += data.effect_value
        "block":
            GameManager.player_data.add_block(data.effect_value)
        "heal":
            GameManager.player_data.heal(data.effect_value)

func _execute_status():
    # 施加状态
    if data.target == "self":
        BattleManager.add_effect("player", data.effect_id, data.effect_stacks)
    else:
        for enemy in BattleManager.enemies:
            BattleManager.add_effect(enemy.id, data.effect_id, data.effect_stacks)

func _execute_core():
    # 核心卡特殊效果
    # 由各流派专属逻辑处理
    pass

func get_display_name() -> String:
    var name_key = data.name_key
    if tier > 1:
        name_key += "_tier%d" % tier
    return tr(name_key)

func get_display_description() -> String:
    var desc_key = data.desc_key
    if tier > 1:
        desc_key += "_tier%d" % tier
    return tr(desc_key)
```

### 五、角色系统 (Character)

**职责**：角色数据、属性计算、技能管理

```gdscript
# scripts/characters/Character.gd
extends Node
class_name Character

signal hp_changed(current: int, max: int)
signal block_changed(amount: int)
signal energy_changed(current: int, max: int)

var id: String
var name_key: String
var desc_key: String

# 基础属性
var base_max_hp: int = 100
var base_energy: int = 3
var base_crit_rate: float = 0.05

# 当前状态
var current_hp: int = 100
var max_hp: int = 100
var current_energy: int = 3
var max_energy: int = 3
var block: int = 0

# 流派信息
var current_flow: String = ""
var flow_levels: Dictionary = {}  # 流派等级

#  Buff/Debuff
var buffs: Dictionary = {}
var debuffs: Dictionary = {}

func setup(character_data: CharacterData):
    id = character_data.id
    name_key = character_data.name_key
    desc_key = character_data.desc_key
    base_max_hp = character_data.base_hp
    base_energy = character_data.base_energy
    base_crit_rate = character_data.base_crit_rate
    current_flow = character_data.initial_flow
    
    max_hp = base_max_hp
    current_hp = max_hp
    max_energy = base_energy
    current_energy = max_energy

func take_damage(amount: int, element: String = "") -> int:
    # 计算实际伤害（考虑格挡和防御）
    var actual_damage = _calculate_actual_damage(amount, element)
    
    # 扣除格挡
    if block > 0:
        var block_damage = min(block, actual_damage)
        block -= block_damage
        actual_damage -= block_damage
        block_changed.emit(block)
    
    # 扣除生命
    current_hp = max(0, current_hp - actual_damage)
    hp_changed.emit(current_hp, max_hp)
    
    # 检查死亡
    if current_hp <= 0:
        _on_death()
    
    return actual_damage

func _calculate_actual_damage(base_damage: int, element: String) -> int:
    var damage = base_damage
    
    # 元素抗性
    var resistance = get_element_resistance(element)
    damage = int(damage * (1 - resistance))
    
    # 状态修正
    damage = BattleManager.calculate_damage_received_modifier(damage)
    
    return max(1, damage)

func heal(amount: int) -> int:
    var old_hp = current_hp
    current_hp = min(max_hp, current_hp + amount)
    var healed = current_hp - old_hp
    hp_changed.emit(current_hp, max_hp)
    return healed

func add_block(amount: int):
    block += amount
    block_changed.emit(block)

func get_crit_rate() -> float:
    var rate = base_crit_rate
    
    # 流派加成
    rate += _get_flow_crit_bonus()
    
    # Buff 加成
    if "crit_up" in buffs:
        rate += buffs["crit_up"]
    
    return min(1.0, rate)

func get_crit_multiplier() -> float:
    var mult = 1.5
    
    # 词条加成
    if "deadly" in GameManager.player_data.affixes:
        mult += 0.5
    
    return mult

func _get_flow_crit_bonus() -> float:
    # 根据当前流派返回暴击加成
    match current_flow:
        "shooting":
            return 0.15  # 游侠基础暴击高
        "stealth":
```

```gdscript
        var save_data = file.get_var()
        file.close()
        save_loaded.emit(slot)
        return save_data
    
    return {}

    var file_path = SAVE_PATH + SAVE_FILE_PREFIX + str(slot) + ".save"

        DirAccess.remove_absolute(file_path)
        save_deleted.emit(slot)
        return true
    
    return false
```

```gdscript
func _load_csv(file_path: String) -> Dictionary:
    var result = {}
    var file = FileAccess.open(file_path, FileAccess.READ)
    if file:
        # 跳过标题行
        file.get_line()
        
        while not file.eof_reached():
            var line = file.get_line()
            var parts = line.split(",", false, 1)
            if parts.size() == 2:
                result[parts[0]] = parts[1]
        
        file.close()
    
    return result

    if locale in supported_locales:

        current_locale = locale
        TranslationServer.set_locale(locale)
        language_changed.emit(locale)

    # 替换参数
    for arg_key in args:
        text = text.replace("{" + arg_key + "}", str(args[arg_key]))
    
    return text
```

```gdscript
class_name ObjectPool

var pool: Array[Node] = []
var scene: PackedScene

func _init(scene_to_pool: PackedScene, initial_size: int = 10):
    scene = scene_to_pool
    for i in range(initial_size):
        var instance = scene.instantiate()
        instance.set_process(false)
        instance.visible = false
        add_child(instance)
        pool.append(instance)

func get() -> Node:
    for instance in pool:
        if not instance.is_processing():
            instance.set_process(true)
            instance.visible = true
            return instance
    
    # 池为空，创建新实例
    var instance = scene.instantiate()
    add_child(instance)
    pool.append(instance)
    return instance

func return_to_pool(instance: Node):
    instance.set_process(false)
    instance.visible = false

func clear():
    for instance in pool:
        instance.queue_free()
    pool.clear()
```

## 一、事件系统概述

### 1. 设计目标

事件系统是远征玩法中**随机性**的核心来源，为每局游戏提供不同的叙事体验和策略选择。

**核心循环**：
```
到达事件节点 → 触发事件 → 做出选择 → 获得结果 → 继续远征
```

### 2. 事件分类

| 分类 | 占比 | 说明 |
|:---|:---:|:---|
| **奖励事件** | 40% | 获得资源、卡牌、增益 |
| **中性事件** | 30% | 选择代价与回报 |
| **负面事件** | 30% | 损失、战斗、诅咒 |

### 3. 事件数据结构

```gdscript
# EventResource.gd
class_name EventResource extends Resource

@export var event_id: String
@export var event_name: String
@export var event_type: String  # reward, neutral, negative
@export var description: String
@export var choices: Array[EventChoice]
@export var requirements: Dictionary  # 触发条件
@export var weight: int = 100  # 随机权重
@export var min_difficulty: int = 1  # 最低难度
@export var environments: Array[String]  # 适用环境
```

## 六、事件触发规则

### 1. 权重系统

```gdscript
# 事件权重计算
func calculate_event_weight(event: EventResource) -> int:
    var weight = event.weight
    
    # 难度修正
    if event.min_difficulty > current_difficulty:
        weight = 0
    
    # 环境修正
    if not event.environments.is_empty():
        if current_environment not in event.environments:
            weight *= 0.5
    
    # 已触发修正 (避免重复)
    if event.event_id in triggered_events:
        weight *= 0.1
    
    return weight
```

### 2. 触发条件

| 条件类型 | 说明 | 示例 |
|:---|:---|:---|
| 难度要求 | 最低/最高难度 | min_difficulty >= 2 |
| 环境要求 | 特定环境触发 | environments = ["rain", "storm"] |
| 前置事件 | 需要先触发某事件 | requires = ["event_001"] |
| 资源要求 | 需要特定资源 | requires_gold >= 50 |
| 流派要求 | 需要特定流派卡牌 | requires_card_type = "thunder" |

### 3. 保底机制

- 每局至少触发 2 个奖励事件
- 连续 3 个负面事件后强制奖励事件
- BOSS 前必有休息事件

## 八、Godot 实现方案

### 1. 事件管理器

```gdscript
# EventManager.gd - 单例
class_name EventManager extends Node

signal event_triggered(event_id: String)
signal event_completed(event_id: String, choice_index: int)

var event_pool: Array[EventResource] = []
var triggered_events: Array[String] = []
var current_event: EventResource = null

func load_events():
    # 从配置文件加载所有事件
    var event_files = list_event_files()
    for file in event_files:
        var event = load(file) as EventResource
        event_pool.append(event)

func trigger_random_event(node_type: String, environment: String) -> EventResource:
    var available_events = get_available_events(node_type, environment)
    if available_events.is_empty():
        return null
    
    # 权重随机
    var weights = []
    for event in available_events:
        weights.append(calculate_weight(event))
    
    var selected = weighted_random(available_events, weights)
    current_event = selected
    triggered_events.append(selected.event_id)
    event_triggered.emit(selected.event_id)
    
    return selected

func get_available_events(node_type: String, environment: String) -> Array:
    var available = []
    for event in event_pool:
        if is_event_available(event, node_type, environment):
            available.append(event)
    return available

func is_event_available(event: EventResource, node_type: String, environment: String) -> bool:
    # 检查难度
    if event.min_difficulty > current_difficulty:
        return false
    
    # 检查环境
    if not event.environments.is_empty():
        if environment not in event.environments:
            return false
    
    # 检查是否已触发
    if event.event_id in triggered_events:
        return false
    
    return true
```

### 2. 事件选择处理

```gdscript
# EventUI.gd
func on_choice_selected(choice_index: int):
    var event = EventManager.current_event
    var choice = event.choices[choice_index]
    
    # 执行选择结果
    for effect in choice.effects:
        execute_effect(effect)
    
    EventManager.event_completed.emit(event.event_id, choice_index)
    
    # 关闭事件界面，返回地图
    close_event_ui()
```

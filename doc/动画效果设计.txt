## 一、动画系统概述

### 1. 动画分类

| 分类 | 实现方式 | 数量 | 说明 |
|:---|:---|:---:|:---|
| 角色动画 | 精灵图序列帧 | 20+ | 玩家/敌人/BOSS 的动作 |
| 战斗特效 | 精灵图 + 粒子 | 15+ | 打击/元素/状态效果 |
| 机制杀特效 | 精灵图+Shader+ 粒子 | 6 | 全屏演出效果 |
| UI 动画 | 纯代码 (Tween) | 10+ | 界面交互动画 |
| 摄像机效果 | 代码 | 5 | 震动/缩放/跟随 |

### 2. 动画优先级

| 优先级 | 动画类型 | 可打断性 | 说明 |
|:---:|:---|:---|:---|
| 1 (最高) | 死亡动画 | ❌ 不可打断 | 播放时必须完成 |
| 2 | 受击动画 | ⚠️ 仅死亡可打断 | 受击时无敌 |
| 3 | 攻击动画 | ⚠️ 受击/死亡可打断 | 攻击前摇可取消 |
| 4 | 防御/格挡 | ⚠️ 受击/死亡可打断 | 持续状态 |
| 5 (最低) | 待机/移动 | ✅ 可被任何动画打断 | 基础状态 |

### 3. 动画帧率规范

| 动画类型 | 帧率 | 说明 |
|:---|:---:|:---|
| 角色动画 | 30 FPS | 平衡流畅度与资源 |
| 战斗特效 | 30-60 FPS | 高速动画用 60FPS |
| 机制杀特效 | 30 FPS | 演出效果优先 |
| UI 动画 | 60 FPS | 使用 Tween 实现 |

### 3. 雷电特效 (fx_thunder)

**基本信息**：
- 实现方式：Line2D + Shader + 粒子
- 持续时间：0.5 秒
- 颜色：蓝白色 (#A0D8FF)

**实现步骤**：

```gdscript
# ThunderEffect.gd
class_name ThunderEffect extends Node2D

@export var start_point: Vector2
@export var end_point: Vector2
@export var bolt_count: int = 3

var line_nodes: Array[Line2D] = []

func _ready():
    create_lightning_bolts()
    play()

func create_lightning_bolts():
    for i in range(bolt_count):
        var line = Line2D.new()
        line.width = 4.0
        line.default_color = Color(0.6, 0.8, 1.0)
        
        # 生成闪电路径（随机折线）
        var points = PackedVector2Array()
        var current = start_point
        points.append(current)
        
        var segments = 8
        for j in range(segments):
            var t = float(j + 1) / segments
            var target = lerp(start_point, end_point, t)
            # 添加随机偏移
            target.x += randf_range(-20, 20)
            target.y += randf_range(-20, 20)
            points.append(target)
        
        line.points = points
        add_child(line)
        line_nodes.append(line)

func play():
    var tween = create_tween()
    
    # 闪电动画
    for line in line_nodes:
        tween.parallel().tween_property(line, "width", 8.0, 0.05)
        tween.parallel().tween_property(line, "modulate:a", 1.0, 0.05)
    
    # 保持
    await get_tree().create_timer(0.1).timeout
    
    # 淡出
    tween = create_tween()
    for line in line_nodes:
        tween.parallel().tween_property(line, "modulate:a", 0.0, 0.3)
    
    tween.tween_callback(queue_free)
```

**Shader 效果**：
```glsl
// lightning.gdshader
shader_type canvas_item;
blend_mode add;

uniform float time : hint_range(0, 1) = 0.0;
uniform vec4 lightning_color : source_color = vec4(0.6, 0.8, 1.0, 1.0);

void fragment() {
    vec4 tex = texture(TEXTURE, UV);
    float flicker = sin(time * 20.0) * 0.3 + 0.7;
    COLOR = vec4(lightning_color.rgb, tex.a) * flicker;
}
```

### 4. 火焰特效 (fx_fire)

**基本信息**：
- 实现方式：Particle2D
- 持续时间：1.0 秒（可持续）
- 颜色：黄→橙→红渐变

**粒子参数**：
```gdscript
# FireEffect.gd
class_name FireEffect extends GPUParticles2D

func _ready():
    var material = ParticleProcessMaterial.new()
    
    # 发射参数
    material.emission_shape = ParticleProcessMaterial.EMISSION_SHAPE_POINT
    material.emission_point_offset = Vector3(0, 0, 0)
    material.amount = 50
    material.lifetime = 0.8
    material.lifetime_randomness = 0.3
    
    # 速度参数
    material.initial_velocity_min = 30.0
    material.initial_velocity_max = 60.0
    material.velocity_scale = 1.0
    
    # 重力
    material.gravity = Vector3(0, -10, 0)
    
    # 粒子大小
    material.scale_min = 0.5
    material.scale_max = 1.5
    material.scale_curve = create_scale_curve()
    
    # 颜色渐变
    material.color = Color(1.0, 0.8, 0.2)  # 黄色
    process_material = material

func create_scale_curve():
    var curve = Curve.new()
    curve.add_point(0, 0.5)
    curve.add_point(0.3, 1.0)
    curve.add_point(1.0, 0.0)
    return curve
```

### 5. 冰冻特效 (fx_freeze)

**基本信息**：
- 实现方式：Shader + 粒子
- 持续时间：0.8 秒
- 颜色：蓝白色

**实现步骤**：
```gdscript
# FreezeEffect.gd
class_name FreezeEffect extends Node2D

func apply_freeze(target: Node2D):
    # 添加冰霜 Shader
    var shader_mat = ShaderMaterial.new()
    shader_mat.shader = preload("res://shaders/freeze.gdshader")
    target.material = shader_mat
    
    # 生成冰晶粒子
    var particles = ice_crystal_scene.instantiate()
    particles.global_position = target.global_position
    get_tree().current_scene.add_child(particles)
    
    # Shader 动画
    var tween = create_tween()
    tween.tween_property(shader_mat, "shader_parameter/freeze_amount", 1.0, 0.5)
    
    # 解除冰冻
    await get_tree().create_timer(0.8).timeout
    tween = create_tween()
    tween.tween_property(shader_mat, "shader_parameter/freeze_amount", 0.0, 0.3)
    tween.tween_callback(func(): target.material = null)
```

**冰冻 Shader**：
```glsl
// freeze.gdshader
shader_type canvas_item;

uniform float freeze_amount : hint_range(0, 1) = 0.0;
uniform vec4 ice_color : source_color = vec4(0.5, 0.8, 1.0, 0.5);

void fragment() {
    vec4 tex_color = texture(TEXTURE, UV);
    COLOR = mix(tex_color, ice_color, freeze_amount * 0.5);
    
    // 添加冰晶效果
    float ice_pattern = sin(UV.x * 50.0) * sin(UV.y * 50.0);
    COLOR.rgb += ice_pattern * freeze_amount * 0.2;
}
```

```gdscript
var shake_duration: float = 0.0
var shake_intensity: float = 0.0
var decay_rate: float = 5.0

# 震动类型
enum ShakeType {
    LIGHT,      # 轻微震动（受击）
    MEDIUM,     # 中等震动（暴击）
    HEAVY,      # 强烈震动（爆炸）
    CUSTOM      # 自定义

const SHAKE_PRESETS = {
    ShakeType.LIGHT: {"intensity": 3, "duration": 0.2},
    ShakeType.MEDIUM: {"intensity": 6, "duration": 0.3},
    ShakeType.HEAVY: {"intensity": 12, "duration": 0.5},
}

func _process(delta: float):
    if shake_duration > 0:
        # 计算当前强度（带衰减）
        var current_intensity = shake_intensity * (shake_duration / get_total_duration())
        
        # 应用震动
        offset = Vector2(
            randf_range(-current_intensity, current_intensity),
            randf_range(-current_intensity, current_intensity)
        )
        
        shake_duration -= delta
    else:
        offset = Vector2.ZERO

func shake(intensity: float, duration: float):
    """应用屏幕震动"""
    shake_intensity = max(shake_intensity, intensity)
    shake_duration = max(shake_duration, duration)

func shake_preset(shake_type: ShakeType):
    """使用预设震动"""
    if shake_type in SHAKE_PRESETS:
        var preset = SHAKE_PRESETS[shake_type]
        shake(preset.intensity, preset.duration)

func get_total_duration() -> float:
    return shake_duration

# 静态方法，方便全局调用
static func shake_camera(camera: Camera2D, intensity: float, duration: float):
    if camera is CameraShake:
        camera.shake(intensity, duration)

static func shake_preset_camera(camera: Camera2D, shake_type: ShakeType):
```

```gdscript
# CharacterAnimationController.gd
class_name CharacterAnimationController extends Node

enum State {
    IDLE,       # 待机
    MOVE,       # 移动
    ATTACK,     # 攻击
    HIT,        # 受击
    DEFEND,     # 防御
    DEATH       # 死亡
}

# 优先级定义
const PRIORITY = {
    State.DEATH: 1,
    State.HIT: 2,
    State.ATTACK: 3,
    State.DEFEND: 4,
    State.MOVE: 5,
    State.IDLE: 6
}

var current_state: State = State.IDLE
var current_priority: int = PRIORITY[State.IDLE]

@onready var anim_player: AnimationPlayer = $AnimationPlayer
@onready var sprite: AnimatedSprite2D = $AnimatedSprite2D

signal state_changed(new_state: State)
signal animation_frame(frame_name: String)

func _ready():
    anim_player.animation_finished.connect(_on_animation_finished)

func can_play_state(new_state: State) -> bool:
    """检查是否可以播放新状态的动画"""
    return PRIORITY[new_state] <= current_priority

func play_animation(anim_name: String, state: State):
    """播放动画"""
    if not can_play_state(state):
        return false
    
    current_state = state
    current_priority = PRIORITY[state]
    anim_player.play(anim_name)
    state_changed.emit(state)
    return true

func _on_animation_finished(anim_name: String):
    """动画结束回调"""
    match current_state:
        State.ATTACK:
            # 攻击结束后恢复到待机
            transition_to_idle()
        State.HIT:
            # 受击结束后恢复到待机
            transition_to_idle()
        State.DEFEND:
            # 防御结束后恢复到待机
            transition_to_idle()
        State.DEATH:
            # 死亡状态不恢复
            pass

func transition_to_idle():
    """过渡到待机状态"""
    current_state = State.IDLE
    current_priority = PRIORITY[State.IDLE]
    state_changed.emit(State.IDLE)

# 快捷方法
func attack(anim_name: String = "attack"):
    return play_animation(anim_name, State.ATTACK)

func hit(anim_name: String = "hit"):
    return play_animation(anim_name, State.HIT)

func die(anim_name: String = "death"):
    return play_animation(anim_name, State.DEATH)

func idle(anim_name: String = "idle"):
    if can_play_state(State.IDLE):
        return play_animation(anim_name, State.IDLE)
    return false
```

## 十、性能优化

### 1. 对象池系统

**完整实现**：
```gdscript
# EffectPool.gd
class_name EffectPool extends Node

@export var effect_scene: PackedScene
@export var pool_size: int = 20

var available_effects: Array[Node2D] = []
var active_effects: Array[Node2D] = []

func _ready():
    for i in range(pool_size):
        var effect = effect_scene.instantiate()
        effect.visible = false
        effect.set_process(false)
        add_child(effect)
        available_effects.append(effect)

func get_effect() -> Node2D:
    """从池中获取一个效果"""
    if available_effects.is_empty():
        # 池为空时动态创建
        var effect = effect_scene.instantiate()
        effect.visible = false
        add_child(effect)
        return effect
    
    var effect = available_effects.pop_back()
    active_effects.append(effect)
    return effect

func return_effect(effect: Node2D):
    """归还效果到池中"""
    var idx = active_effects.find(effect)
    if idx != -1:
        active_effects.remove_at(idx)
    
    effect.visible = false
    effect.set_process(false)
    available_effects.append(effect)

func play_effect_at(position: Vector2, auto_return: bool = true, return_delay: float = 1.0):
    """在指定位置播放效果"""
    var effect = get_effect()
    effect.global_position = position
    effect.visible = true
    effect.set_process(true)
    
    if effect.has_method("play"):
        effect.play()
    
    if auto_return:
        # 自动归还
        var timer = get_tree().create_timer(return_delay)
        timer.timeout.connect(func(): return_effect(effect))
```

**使用示例**：
```gdscript
# 在场景中
@onready var hit_effect_pool = $HitEffectPool

    hit_effect_pool.play_effect_at(position, true, 0.5)
```

### 3. LOD 系统（细节等级）

```gdscript
# LODController.gd
class_name LODController extends Node

@export var high_detail_sprite: AnimatedSprite2D
@export var low_detail_sprite: AnimatedSprite2D
@export var target: Node2D
@export var camera: Camera2D

var lod_distances = {
    "high": 300,   # 300 像素内使用高细节
    "medium": 600, # 600 像素内使用中细节
    "low": 1000    # 1000 像素外使用低细节
}

func _process(_delta: float):
    if not target or not camera:
        return
    
    var distance = camera.global_position.distance_to(target.global_position)
    update_lod(distance)

func update_lod(distance: float):
    if distance < lod_distances.high:
        set_lod("high")
    elif distance < lod_distances.medium:
        set_lod("medium")
    else:
        set_lod("low")

func set_lod(lod: String):
    match lod:
        "high":
            high_detail_sprite.visible = true
            low_detail_sprite.visible = false
        "medium":
            # 可以降低帧率或简化动画
            pass
        "low":
            high_detail_sprite.visible = false
            low_detail_sprite.visible = true
```

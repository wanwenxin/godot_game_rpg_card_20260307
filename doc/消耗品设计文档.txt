## 一、消耗品系统概述

### 1. 设计理念

消耗品是连接**局外基地**与**局内战斗**的核心纽带。玩家在基地制造消耗品，带入远征，在战斗中消耗使用，形成完整的资源循环。

**核心循环**：
```
基地制造 → 选择带入 → 战斗使用 → 获取资源 → 升级制造
```

### 2. 消耗品分类

| 分类 | 说明 | 使用场景 |
|:---|:---|:---|
| **药剂类** | 恢复/增益型消耗品 | 战斗中使用 |
| **装置类** | 环境交互/战术道具 | 战斗中使用 |
| **特殊类** | 一次性强力效果 | 关键时刻使用 |

### 3. 消耗品卡牌属性

```gdscript
# ConsumableCardResource.gd
class_name ConsumableCardResource extends Resource

@export var consumable_id: String
@export var consumable_name: String
@export var consumable_type: String  # potion, device, special
@export var cost: int = 0  # 能量消耗，多数消耗品为 0
@export var effect_list: Array[EffectData]
@export var max_carry: int = 3  # 最大携带数量
@export var rarity: int = 1  # 稀有度
@export var craft_recipe: Dictionary  # 制造配方
```

## 八、Godot 实现方案

### 1. 消耗品管理器

```gdscript
# ConsumableManager.gd - 单例
class_name ConsumableManager extends Node

signal consumable_used(consumable_id: String)
signal consumable_added(consumable_id: String, count: int)

var inventory: Dictionary = {}  # 消耗品库存
var equipped_consumables: Array[String] = []  # 带入的消耗品
var max_carry_weight: int = 10

func use_consumable(consumable_id: String, target: Node = null) -> bool:
    if not inventory.has(consumable_id) or inventory[consumable_id] <= 0:
        return false
    
    var consumable_data = get_consumable_data(consumable_id)
    execute_effect(consumable_data.effect_list, target)
    
    inventory[consumable_id] -= 1
    consumable_used.emit(consumable_id)
    
    if inventory[consumable_id] <= 0:
        inventory.erase(consumable_id)
    
    return true

func add_consumable(consumable_id: String, count: int = 1):
    if inventory.has(consumable_id):
        inventory[consumable_id] += count
    else:
        inventory[consumable_id] = count
    consumable_added.emit(consumable_id, count)

func get_carry_weight() -> int:
    var weight = 0
    for consumable_id in equipped_consumables:
        var data = get_consumable_data(consumable_id)
        weight += data.get("carry_weight", 1)
    return weight
```

### 2. 消耗品效果执行

```gdscript
# ConsumableEffect.gd
class_name ConsumableEffect extends RefCounted

var effect_type: String
var value: int
var duration: int

func execute(target: Node, source: Node = null):
    match effect_type:
        "heal":
            target.heal(value)
        "damage":
            target.take_damage(value)
        "buff":
            target.add_status(effect_type, value, duration)
        "draw_card":
            CardManager.draw_cards(value)
        "energy_gain":
            EnergyManager.add_energy(value)
```

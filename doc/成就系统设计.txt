## 一、成就系统概述

### 1. 设计目标

成就系统为玩家提供**长期目标**和**收集要素**，增加游戏耐玩性和重复游玩价值。

**核心功能**：
- 记录玩家游戏历程
- 提供挑战性目标
- 解锁奖励和展示内容
- 增加游戏沉浸感

### 2. 成就分类

| 分类 | 占比 | 说明 |
|:---|:---:|:---|
| **进度成就** | 25% | 游戏进程相关 |
| **挑战成就** | 30% | 高难度挑战 |
| **收集成就** | 25% | 卡牌/资源收集 |
| **机制杀成就** | 20% | 特殊击杀方式 |

### 3. 成就数据结构

```gdscript
# AchievementResource.gd
class_name AchievementResource extends Resource

@export var achievement_id: String
@export var achievement_name: String
@export var description: String
@export var category: String  # progress, challenge, collection, mechanic
@export var condition_type: String  # count, reach, defeat, collect
@export var condition_value: int
@export var reward_type: String  # card, resource, display, buff
@export var reward_data: Dictionary
@export var is_hidden: bool = false
@export var icon_path: String
```

## 八、Godot 实现方案

### 1. 成就管理器

```gdscript
# AchievementManager.gd - 单例
class_name AchievementManager extends Node

signal achievement_unlocked(achievement_id: String)
signal achievement_progress_updated(achievement_id: String, current: int, target: int)

var achievements: Dictionary = {}  # 已解锁成就
var progress: Dictionary = {}  # 当前进度
var achievement_resources: Array[AchievementResource] = []

func _ready():
    load_achievements()
    load_progress()

func unlock_achievement(achievement_id: String):
    if achievements.has(achievement_id):
        return  # 已解锁
    
    var achievement = get_achievement_resource(achievement_id)
    achievements[achievement_id] = true
    
    # 发放奖励
    give_reward(achievement.reward_type, achievement.reward_data)
    
    # 保存进度
    save_progress()
    
    # 通知 UI
    achievement_unlocked.emit(achievement_id)
    show_achievement_popup(achievement)

func update_progress(achievement_id: String, amount: int = 1):
    var achievement = get_achievement_resource(achievement_id)
    if not achievement or achievements.has(achievement_id):
        return
    
    if not progress.has(achievement_id):
        progress[achievement_id] = 0
    
    progress[achievement_id] += amount
    
    if progress[achievement_id] >= achievement.condition_value:
        unlock_achievement(achievement_id)
    else:
        achievement_progress_updated.emit(
            achievement_id, 
            progress[achievement_id], 
            achievement.condition_value
        )

func check_achievements(condition_type: String, data: Dictionary):
    # 检查所有相关成就
    for achievement in achievement_resources:
        if achievement.condition_type == condition_type:
            if check_condition(achievement, data):
                update_progress(achievement.achievement_id)
```

### 2. 成就 UI

```gdscript
# AchievementUI.gd
func show_achievement_popup(achievement: AchievementResource):
    var popup = achievement_popup_scene.instantiate()
    get_tree().current_scene.add_child(popup)
    popup.setup(
        achievement.achievement_name,
        achievement.description,
        achievement.icon_path
    )
    
    # 3 秒后自动消失
    await get_tree().create_timer(3.0).timeout
    popup.queue_free()
```

### 3. 成就保存

```gdscript
# 成就数据保存
func save_progress():
    var save_data = {
        "achievements": achievements,
        "progress": progress
    }
    
    var file = FileAccess.open("user://achievements.save", FileAccess.WRITE)
    file.store_var(save_data)

func load_progress():
    if not FileAccess.file_exists("user://achievements.save"):
        return
    
    var file = FileAccess.open("user://achievements.save", FileAccess.READ)
    var save_data = file.get_var()
    
    achievements = save_data.get("achievements", {})
    progress = save_data.get("progress", {})
```
